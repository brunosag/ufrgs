\section{Introdução}

Neste projeto, desenvolvemos uma versão completa do processador hipotético Neander, utilizando uma unidade de controle baseada em expressões lógicas e uma memória para armazenar o programa e os dados. O sistema foi projetado para operar com um clock de entrada de 50MHz, reduzido para aproximadamente 2Hz para a execução das instruções, e inclui os principais componentes de um processador, como Unidade Lógica e Aritmética (ULA), registradores (AC, PC, RDM, REM, RI) e decodificador de instruções. O projeto foi implementado e simulado com o software Quartus II, com o objetivo de sintetizá-lo e testá-lo em uma FPGA Cyclone III (modelo EP3C16F484C6).

\subsection{Entradas}

\begin{itemize}
    \item \texttt{CLK\_IN}: Clock inicial com frequência de 50MHz.
    \item \texttt{RST}: Sinal de reset assíncrono para todo o sistema.
    \item \texttt{DEBUG}: Habilita execução passo a passo através do botão \texttt{STEP}.
    \item \texttt{STEP}: Botão para avançar um ciclo de clock no modo \texttt{DEBUG}.
\end{itemize}

\subsection{Saídas}

\begin{itemize}
    \item \texttt{HEX3[6..0]}, \texttt{HEX2[6..0]}: Representação dos 8 bits do registrador PC em dois visores de 7 segmentos hexadecimais.
    \item \texttt{HEX1[6..0]}, \texttt{HEX0[6..0]}: Representação dos 8 bits do acumulador AC em dois visores de 7 segmentos hexadecimais.
    \item \texttt{N}: LED para exibir o valor da flag de estado Negativo (N).
    \item \texttt{Z}: LED para exibir o valor da flag de estado Zero (Z).
    \item \texttt{HLT}: LED para indicar que o processador foi parado pela instrução HLT.
\end{itemize}

\subsection{Módulos Principais}

\begin{itemize}
    \item \textit{Unidade Lógica e Aritmética (ULA)}: Realiza as operações lógicas (AND, OR, NOT) e aritméticas (ADD) do processador.
    \item \textit{Registradores}: Unidades de armazenamento de 8 bits (AC, RDM, REM, RI) e um contador de 8 bits (PC) com controle de carga e incremento.
    \item \textit{Unidade de Controle (UC)}: Gera os sinais de controle para os demais componentes a partir das instruções e dos passos do temporizador.
    \item \textit{Memória}: Armazena as instruções do programa e os dados (implementada como ROM ou RAM).
    \item \textit{Decodificador de Instruções}: Decodifica o opcode da instrução atual para a unidade de controle.
    \item \textit{Temporizador}: Contador de 3 bits que sequencia os passos de execução de cada instrução.
\end{itemize}

create_clock -name "main_clk" -period 20.000 -waveform {0.000 10.000} [get_ports CLK_IN]
derive_clock_uncertainty

set_input_delay -clock "main_clk" -max 5.0 [remove_from_collection [all_inputs] [get_ports CLK_IN]]
set_input_delay -clock "main_clk" -min -1.0 [remove_from_collection [all_inputs] [get_ports CLK_IN]]

set_output_delay -clock "main_clk" -max 5.0 [all_outputs]
set_output_delay -clock "main_clk" -min -1.0 [all_outputs]
